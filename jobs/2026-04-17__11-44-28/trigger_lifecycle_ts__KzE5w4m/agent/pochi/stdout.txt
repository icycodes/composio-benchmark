import { Composio } from '@composio/core';

async function main() {
    const composio = new Composio({ apiKey: process.env.COMPOSIO_API_KEY || "test" });
    const userId = "zealt-user01";

    console.log("Connecting to Composio and creating a session for user:", userId);
    const session = await composio.create(userId);
    console.log("Created session for user.");

    // Find active github connected account
    const accounts = await composio.connectedAccounts.list({ userIds: [userId] });
    const githubAccount = accounts.items.find(a => a.toolkit.slug === "github" && a.status === "ACTIVE");
    if (!githubAccount) {
        throw new Error("No active github connected account found for user.");
    }
    console.log("Using connected account ID:", githubAccount.id);

    const types = await composio.triggers.listTypes({ toolkits: ["github"] });
    console.log("Available github trigger types:", types.items.map(t => t.slug));

    const githubTrigger = types.items.find(t => t.slug === "GITHUB_ISSUE_ADDED_EVENT" || t.slug.includes("ISSUE"));
    if (!githubTrigger) {
        throw new Error("Could not find GITHUB_ISSUE_ADDED_EVENT or similar trigger type.");
    }
    console.log("Found trigger type:", githubTrigger.slug);

    // Creates an active trigger
    const trigger = await composio.triggers.create(userId, githubTrigger.slug, {
        connectedAccountId: githubAccount.id,
        triggerConfig: { owner: "test-owner", repo: "test-repo" }
    });
    console.log("Created trigger with ID:", trigger.triggerId);

    // Disables the newly created trigger.
    const disabled = await composio.triggers.disable(trigger.triggerId);
    console.log("Disabled trigger:", disabled);

    let activeTriggers: any[] = [];
    const active = await composio.triggers.listActive({ connectedAccountIds: [githubAccount.id] });
    activeTriggers = active.items;
    
    console.log("Active triggers count:", activeTriggers.length);
    const isStillActive = activeTriggers.some(t => t.id === trigger.triggerId);
    console.log("Is trigger still active?", isStillActive);

    const statusData = {
        triggerId: trigger.triggerId,
        slug: githubTrigger.slug,
        disabledResult: disabled,
        isStillActive: isStillActive,
        finalStatus: "disabled"
    };

    fs.writeFileSync("/home/user/trigger_status.json", JSON.stringify(statusData, null, 2));
    console.log("Saved status to /home/user/trigger_status.json");
}
