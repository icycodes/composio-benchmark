import os
import json
from openai import OpenAI
from composio import Composio

# Initialize clients
client = Composio()
openai_client = OpenAI()

# Get tools
tools = client.tools.get(user_id="zealt-user01", tools=["COMPOSIO_SEARCH_TOOLS", "COMPOSIO_MULTI_EXECUTE_TOOL"])

# Trial ID and repo
trial_id = "tool_discovery_agent_python__Ar7HoCo"
repo_name = f"zealt-user01/discovery-test-{trial_id}"

messages = [
    {"role": "system", "content": "You are a helpful assistant. Use the provided tools to search for and execute tasks. ALWAYS use COMPOSIO_MULTI_EXECUTE_TOOL to execute the tools you find."},
    {"role": "user", "content": f"First, search for a tool to star a github repository. Then, use the tool you found to star the repository '{repo_name}'."}
]

with open("/home/user/discovery.log", "w") as log_file:
    def log(msg):
        print(msg)
        log_file.write(msg + "\n")
        log_file.flush()

    log("Starting agent loop...")
    
    while True:
        response = openai_client.chat.completions.create(
            model="gpt-4o",
            messages=messages,
            tools=tools,
            tool_choice="auto"
        )
        
        msg = response.choices[0].message
        messages.append(msg)
        
        if msg.tool_calls:
            for tool_call in msg.tool_calls:
                name = tool_call.function.name
                args = json.loads(tool_call.function.arguments)
                log(f"Executing tool {name} with arguments {args}")
                
                try:
                    res = client.tools.execute(
                        slug=name,
                        arguments=args,
                        user_id="zealt-user01",
                        dangerously_skip_version_check=True
                    )
                    log(f"Tool {name} result: {json.dumps(res)[:500]}...")
                    
                    messages.append({
                        "role": "tool",
                        "tool_call_id": tool_call.id,
                        "name": name,
                        "content": json.dumps(res)
                    })
                except Exception as e:
                    log(f"Tool execution failed: {e}")
                    messages.append({
                        "role": "tool",
                        "tool_call_id": tool_call.id,
                        "name": name,
                        "content": json.dumps({"error": str(e)})
                    })
        else:
            log(f"Agent finished: {msg.content}")
            break
