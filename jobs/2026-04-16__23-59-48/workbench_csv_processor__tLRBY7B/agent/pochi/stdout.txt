from composio import Composio

# Read local CSV content
with open('/home/user/data.csv', 'r') as f:
    csv_content = f.read()

composio = Composio()
session = composio.tool_router.create(user_id="zealt-user01")

# Upload CSV content to remote sandbox
upload_code = f"""
from pathlib import Path
csv_content = {csv_content!r}
Path('/home/user/data.csv').write_text(csv_content)
"""
resp = session.execute("COMPOSIO_REMOTE_WORKBENCH", arguments={
    "code_to_execute": upload_code,
    "thought": "Upload CSV content to the workbench sandbox so it can be processed."
})
if resp.error:
    raise SystemExit(f"Upload error: {resp.error}")

# Process CSV in remote sandbox and write result.txt
process_code = """
import pandas as pd

df = pd.read_csv('/home/user/data.csv')
avg_age = df['age'].mean()
with open('/home/user/result.txt', 'w') as f:
    f.write(str(avg_age))
"""
resp = session.execute("COMPOSIO_REMOTE_WORKBENCH", arguments={
    "code_to_execute": process_code,
    "thought": "Compute average age from CSV and write result.txt in sandbox."
})
if resp.error:
    raise SystemExit(f"Process error: {resp.error}")

# Read result.txt from sandbox
resp = session.execute("COMPOSIO_REMOTE_BASH_TOOL", arguments={
    "command": "cat /home/user/result.txt"
})
if resp.error:
    raise SystemExit(f"Download error: {resp.error}")

print(resp.data)
