import os
import sys

with open('/logs/trial_id', 'r') as f:
    trial_id = f.read().strip()

repo_name = f"star-test-{trial_id}"
user_id = "zealt-user01"

from composio_openai import ComposioToolSet, App
from openai import OpenAI
import json

client = OpenAI()
toolset = ComposioToolSet(entity_id=user_id)

tools = toolset.get_tools(apps=[App.GITHUB])

messages = [
    {"role": "system", "content": "You are a helpful assistant that creates and stars GitHub repositories."},
    {"role": "user", "content": f"Create a public repository named '{repo_name}' under the user '{user_id}'. Then, star the newly created repository."}
]

with open('/home/user/agent.log', 'w') as log_file:
    log_file.write(f"Starting agent task for repo: {repo_name} with user {user_id}\n")
    
    while True:
        response = client.chat.completions.create(
            model="gpt-4o",
            messages=messages,
            tools=tools,
            tool_choice="auto",
        )
        
        message = response.choices[0].message
        messages.append(message)
        
        if message.tool_calls:
            log_file.write(f"Tool calls: {[tc.function.name for tc in message.tool_calls]}\n")
            tool_responses = toolset.execute_tool_calls(message.tool_calls, entity_id=user_id)
            for tool_call, tool_response in zip(message.tool_calls, tool_responses):
                messages.append(
                    {
                        "role": "tool",
                        "tool_call_id": tool_call.id,
                        "name": tool_call.function.name,
                        "content": str(tool_response),
                    }
                )
                log_file.write(f"Tool response for {tool_call.function.name}: {tool_response}\n")
        else:
            log_file.write(f"Agent response: {message.content}\n")
            break

import os
import json
from composio import Composio
from composio_openai import OpenAIProvider
from openai import OpenAI

with open('/logs/trial_id', 'r') as f:
    trial_id = f.read().strip()

repo_name = f"star-test-{trial_id}"
user_id = "zealt-user01"

provider = OpenAIProvider()
composio_client = Composio(provider=provider)
openai_client = OpenAI()

tools = composio_client.tools.get(user_id=user_id, toolkits=["github"])

messages = [
    {"role": "system", "content": "You are a helpful assistant that creates and stars GitHub repositories."},
    {"role": "user", "content": f"Create a public repository named '{repo_name}' under the user '{user_id}'. Then, star the newly created repository."}
]

with open('/home/user/agent.log', 'w') as log_file:
    log_file.write(f"Starting agent task for repo: {repo_name} with user {user_id}\n")
    
    while True:
        response = openai_client.chat.completions.create(
            model="gpt-4o",
            messages=messages,
            tools=tools,
            tool_choice="auto",
        )
        
        message = response.choices[0].message
        messages.append(message)
        
        if message.tool_calls:
            log_file.write(f"Tool calls: {[tc.function.name for tc in message.tool_calls]}\n")
            tool_responses = provider.handle_tool_calls(user_id=user_id, response=response)
            for tool_call, tool_response in zip(message.tool_calls, tool_responses):
                messages.append(
                    {
                        "role": "tool",
                        "tool_call_id": tool_call.id,

                        "content": json.dumps(tool_response),
                    }
                )
                log_file.write(f"Tool response for {tool_call.function.name}: {tool_response}\n")
        else:
            log_file.write(f"Agent response: {message.content}\n")
