import os
from composio import Composio

def morning_sweep():
    api_key = os.getenv("COMPOSIO_API_KEY")
    if not api_key:
        print("COMPOSIO_API_KEY not set")
        return

    composio = Composio(api_key=api_key)
    user_id = "zealt-user01"
    report_path = "/home/user/morning_sweep.txt"

    report_lines = []
    report_lines.append(f"Morning Sweep Report for {user_id}")
    report_lines.append("=" * 40)

    try:
        # 1. List all connected accounts for GitHub for zealt-user01
        accounts = composio.connected_accounts.list(user_ids=[user_id], toolkit_slugs=["github"])
        
        if not accounts.items:
            report_lines.append("No connected GitHub accounts found.")
        else:
            for account in accounts.items:
                account_id = account.id
                report_lines.append(f"\nAccount ID: {account_id}")
                report_lines.append("-" * 20)
                
                if account.status != "ACTIVE":
                    report_lines.append(f"Status: {account.status} (Skipping...)")
                    continue

                # 2. For each account, fetch the last 3 issues from any repository
                # We'll use GITHUB_SEARCH_ISSUES or similar if available, 
                # or just GITHUB_LIST_ISSUES_FOR_AUTHENTICATED_USER
                try:
                    # Fetching issues for the specific account
                    # GITHUB_LIST_ISSUES_FOR_AUTHENTICATED_USER is a good candidate
                    # It returns issues assigned to or created by the user across all repos
                    result = composio.tools.execute(
                        slug="github_list_issues_for_authenticated_user",
                        arguments={
                            "per_page": 3,
                            "state": "all",
                            "sort": "created",
                            "direction": "desc"
                        },
                        connected_account_id=account_id
                    )
                    
                    if result and isinstance(result, dict) and 'data' in result:
                        issues = result['data']
                        if not issues:
                            report_lines.append("No issues found for this account.")
                        else:
                            for issue in issues[:3]:
                                title = issue.get('title', 'No Title')
                                state = issue.get('state', 'unknown')
                                url = issue.get('html_url', 'No URL')
                                report_lines.append(f"- [{state.upper()}] {title}")
                                report_lines.append(f"  URL: {url}")
                    else:
                        report_lines.append(f"Could not fetch issues: {result}")
                except Exception as e:
                    report_lines.append(f"Error fetching issues for account {account_id}: {e}")

        # 3. Combine the results into a single summary report
        report_content = "\n".join(report_lines)
        
        # 4. Save the report to /home/user/morning_sweep.txt
        with open(report_path, "w") as f:
            f.write(report_content)
        
        print(f"Report saved to {report_path}")
        print(report_content)

    except Exception as e:
        print(f"An error occurred during the morning sweep: {e}")

if __name__ == "__main__":
    morning_sweep()

def morning_sweep():

        return

    composio = Composio(api_key=api_key)
    user_id = "zealt-user01"
    report_path = "/home/user/morning_sweep.txt"

    report_lines = []
    report_lines.append(f"Morning Sweep Report for {user_id}")
    report_lines.append("=" * 40)

    try:

        accounts = composio.connected_accounts.list(user_ids=[user_id], toolkit_slugs=["github"])

                    continue

                # 2. For each account, fetch the last 3 issues from any repository
                # Since the tool list is restricted, we use the proxy call to fetch issues directly from GitHub API
                try:
                    # Using proxy to call /issues which returns issues for the authenticated user
                    # across all repositories they have access to.
                    result = composio.tools.proxy(
                        method="GET",
                        endpoint="/issues",
                        parameters=[
                            {"name": "per_page", "value": "3", "type": "query"},
                            {"name": "state", "value": "all", "type": "query"},
                            {"name": "sort", "value": "created", "type": "query"},
                            {"name": "direction", "value": "desc", "type": "query"}
                        ],
                        connected_account_id=account_id
                    )

                        issues = result['data']
                        if not issues:
                            report_lines.append("No issues found for this account.")

                        report_lines.append(f"Could not fetch issues via proxy: {result}")
