# Execute code in the workbench to create CSV and calculate average age
code = """
import csv
import os

# Create the CSV file in the sandbox
csv_data = [
    ['name', 'age'],
    ['Alice', 25],
    ['Bob', 30],
    ['Charlie', 35],
    ['Diana', 28],
    ['Eve', 32]
]

# Write CSV to file
with open('/home/user/data.csv', 'w', newline='') as f:
    writer = csv.writer(f)
    writer.writerows(csv_data)

print("CSV file created successfully")

# Read the CSV and calculate average age
with open('/home/user/data.csv', 'r') as f:
    reader = csv.DictReader(f)
    ages = [int(row['age']) for row in reader]
    average_age = sum(ages) / len(ages)

print(f"Average age: {average_age}")

# Save result to result.txt
with open('/home/user/result.txt', 'w') as f:
    f.write(f"Average age: {average_age}\\n")

print("Result saved to result.txt")

# Return the average age
average_age
"""

    'COMPOSIO_REMOTE_WORKBENCH',

)
