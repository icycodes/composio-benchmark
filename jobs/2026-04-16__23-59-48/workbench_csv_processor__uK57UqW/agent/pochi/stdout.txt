from composio import Composio
import os

client = Composio(api_key=os.environ.get("COMPOSIO_API_KEY", "test"))
session = client.get_session(user_id="zealt-user01")
workbench = session.workbench

# 1. Upload data.csv to the remote sandbox.
workbench.upload_file("/home/user/data.csv", "data.csv")

# 2. Execute Python code in the sandbox to calculate the average age.
code = """
import csv
with open('data.csv', 'r') as f:
    reader = csv.DictReader(f)
    ages = [int(row['age']) for row in reader]
    avg = sum(ages) / len(ages)
with open('result.txt', 'w') as f:
    f.write(str(avg))
"""
result = workbench.execute_code(code=code, language="python")

# 3. Download result.txt from the sandbox to /home/user/result.txt
workbench.download_file("result.txt", "/home/user/result.txt")

# 4. Save the average age to /home/user/avg_age.log
with open("/home/user/result.txt", "r") as f:
    avg = f.read().strip()
with open("/home/user/avg_age.log", "w") as f:
    f.write(avg)
EOF
