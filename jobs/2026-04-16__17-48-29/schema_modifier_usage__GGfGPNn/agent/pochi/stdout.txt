from composio import Composio, schema_modifier

@schema_modifier(tools=["HACKERNEWS_GET_LATEST_POSTS"])
def modify_hackernews_schema(tool_name, toolkit_name, schema):
    print("TOOL:", tool_name)
    print("TOOLKIT:", toolkit_name)
    print("SCHEMA TYPE:", type(schema))
    
    if "properties" in schema.input_parameters and "page" in schema.input_parameters["properties"]:
        schema.input_parameters["properties"].pop("page")
    
    if "required" not in schema.input_parameters:
        schema.input_parameters["required"] = []
        
    if "size" not in schema.input_parameters["required"]:
        schema.input_parameters["required"].append("size")
        
    return schema

client = Composio()
tools = client.tools.get(

    tools=["HACKERNEWS_GET_LATEST_POSTS"],
